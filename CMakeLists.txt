cmake_minimum_required(VERSION 3.15)
project(TraceShield LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB_RECURSE TRACESHIELD_SRC CONFIGURE_DEPENDS src/*.cpp)
add_library(traceshield STATIC ${TRACESHIELD_SRC})
target_include_directories(traceshield PUBLIC include)
foreach(tool packetscan iocmatch rulecheck policyaudit)
  add_executable(${tool} tools/${tool}.cpp)
  target_link_libraries(${tool} PRIVATE traceshield)
endforeach()
enable_testing()
foreach(test_name core_tests packet_tests ioc_tests rules_tests policy_tests)
  if(test_name STREQUAL "core_tests")
    set(test_path tests/core/core_tests.cpp)
  elseif(test_name STREQUAL "packet_tests")
    set(test_path tests/packet/packet_tests.cpp)
  elseif(test_name STREQUAL "ioc_tests")
    set(test_path tests/ioc/ioc_tests.cpp)
  elseif(test_name STREQUAL "rules_tests")
    set(test_path tests/rules/rules_tests.cpp)
  else()
    set(test_path tests/policy/policy_tests.cpp)
  endif()
  add_executable(${test_name} ${test_path})
  target_link_libraries(${test_name} PRIVATE traceshield)
  add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()
